% =========================================================
% LINKS
% =========================================================

% Arguments: URL, Font Awesome icon name, visible text.
% Leave the icon name empty to show only the text.
\newcommand{\CVLink}[3]{%
  \href{#1}{%
    \ifblank{#2}{}{\faIcon{#2}\ }%
    #3%
  }%
}

% Arguments: URL, Font Awesome icon name.
% If the icon name is empty, the standard link glyph is used.
\newcommand{\CVIconLink}[2]{%
  \ifblank{#1}
    {}
    {%
      \href{#1}{%
        \color{linkblue}%
        \ifblank{#2}{\faIcon{link}}{\faIcon{#2}}%
      }%
    }%
}

\newcommand{\CVEntrySeparator}{\hspace{4pt}$|$\hspace{4pt}}

% =========================================================
% HEADER
% =========================================================

% Arguments: name, contact items.
\newcommand{\CVHeader}[2]{%
  \begin{center}
    \textbf{\Huge #1}\\[\CVHeaderNameGap]
    \small #2
    \\[\CVHeaderAfterSpace]
  \end{center}
}

\newcommand{\CVHeaderItem}[3]{%
  \CVLink{#1}{#2}{\texttt{#3}}%
}

\newcommand{\CVHeaderSeparator}{\hspace{4pt}$|$\hspace{4pt}}

% =========================================================
% SIMPLE TEXT AND LABEL ROWS
% =========================================================

\newcommand{\CVText}[1]{%
  \ifblank{#1}
    {}
    {%
      \begin{itemize}[leftmargin=0in,label={}]
        \footnotesize\item #1
      \end{itemize}%
    }%
}

\newcommand{\CVLabelRows}[1]{%
  \ifblank{#1}
    {}
    {%
      \footnotesize
      \begin{tabularx}{\textwidth}{%
        @{}
        >{\bfseries}l
        @{\hspace{\CVSkillGap}}
        X
        @{}
      }
        #1%
      \end{tabularx}%
      \par
    }%
}

% Arguments: label, value.
\newcommand{\CVLabelRow}[2]{%
  \ifblank{#1#2}
    {}
    {%
      #1: & #2 \\[\CVSkillRowSpace]%
    }%
}

% =========================================================
% UNIVERSAL ENTRIES
% Used by experience, education, and any similarly styled section.
% =========================================================

\newcommand{\CVEntries}[1]{%
  \ifblank{#1}
    {}
    {%
      \begin{itemize}[leftmargin=0in,label={}]
        #1%
      \end{itemize}%
    }%
}

% Arguments:
% 1: Main title
% 2: Date
% 3: Secondary left text (optional)
% 4: Right-side metadata shown before the date (optional)
% 5: URL (optional)
% 6: Font Awesome icon name (optional; defaults to link)
% 7: Main content, usually \CVBulletList{...} (optional)
% 8: Nested content, usually \CVNestedGroup{...}{...} (optional)
%
% Experience and education use this same parent function. They simply
% populate different subsets of its arguments.
\newcommand{\CVEntry}[8]{%
  \ifblank{#1#2#3#4#5#7#8}
    {}
    {%
      \vspace{\CVEntryBeforeSpace}\item
      \begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
        \textbf{#1}%
        \ifblank{#5}{}{\CVEntrySeparator\CVIconLink{#5}{#6}}%
        & {\color{dark-grey}\fontseries{m}\selectfont\footnotesize
            \ifblank{#4}{}{#4}%
            \ifblank{#2}{}{\ifblank{#4}{}{\CVEntrySeparator}#2}}\vspace{1pt}\\
        \textit{#3} & \\
      \end{tabular*}\vspace{\CVEntryAfterSpace}%
      #7%
      #8%
    }%
}

% =========================================================
% ENTRY CONTENT
% =========================================================

\newcommand{\CVBullet}[1]{%
  \item\footnotesize{#1\vspace{0pt}}%
}

\newcommand{\CVBulletList}[1]{%
  \ifblank{#1}
    {}
    {%
      \begin{itemize}
        #1%
      \end{itemize}\vspace{0pt}%
    }%
}

% Arguments: group label, nested entries.
\newcommand{\CVNestedGroup}[2]{%
  \ifblank{#2}
    {}
    {%
      \vspace{2pt}
      \item[]
        \hspace*{\CVNestedIndent}{\small\textbf{#1}}
      \vspace{-2pt}
      #2%
    }%
}

% Arguments:
% 1: Nested project/title
% 2: Date
% 3: Right-side metadata shown before the date (optional)
% 4: URL (optional)
% 5: Font Awesome icon name (optional; defaults to link)
% 6: Content, usually \CVNestedBulletList{...} (optional)
\newcommand{\CVNestedEntry}[6]{%
  \vspace{0pt}\item
  \hspace*{\CVNestedIndent}%
  \begin{tabular*}{\dimexpr\textwidth-\CVNestedIndent\relax}[t]{l@{\extracolsep{\fill}}r}
    {\small\textbf{---\ \ #1}%
      \ifblank{#4}{}{\CVEntrySeparator\CVIconLink{#4}{#5}}}%
    & {\color{dark-grey}\fontseries{m}\selectfont\footnotesize
        \ifblank{#3}{}{#3}%
        \ifblank{#2}{}{\ifblank{#3}{}{\CVEntrySeparator}#2}}\\
  \end{tabular*}\vspace{\CVNestedHeadingAfterSpace}%
  #6%
}

\newcommand{\CVNestedBulletList}[1]{%
  \ifblank{#1}
    {}
    {%
      \begin{itemize}[
        leftmargin=\CVNestedListLeftMargin,
        topsep=2pt,
        itemsep=1pt,
        parsep=0pt,
        partopsep=0pt
      ]
        #1%
      \end{itemize}\vspace{-1pt}%
    }%
}

% Non-bulleted content used when a JSON content item sets bullet=false.
\newcommand{\CVEntryText}[1]{%
  \ifblank{#1}
    {}
    {%
      \par
      \hspace*{0.18in}%
      \begin{minipage}{\dimexpr\linewidth-0.18in\relax}
        \footnotesize #1
      \end{minipage}%
      \par
    }%
}

\newcommand{\CVNestedText}[1]{%
  \ifblank{#1}
    {}
    {%
      \par
      \hspace*{\CVNestedListLeftMargin}%
      \begin{minipage}{\dimexpr\linewidth-\CVNestedListLeftMargin\relax}
        \footnotesize #1
      \end{minipage}%
      \par
    }%
}

% =========================================================
% REFERENCES
% Reference rows use their own full-width layout: identity and relationship
% stay on the left while the phone number or email is flush right.
% =========================================================

\newcommand{\CVReferences}[1]{%
  \ifblank{#1}
    {}
    {%
      \vspace{\CVReferencesBeforeSpace}%
      \begin{itemize}[leftmargin=0in,label={}]
        #1%
      \end{itemize}%
    }%
}

% Arguments: name, position or relationship, phone number or email.
\newcommand{\CVReference}[3]{%
  \ifblank{#1#2#3}
    {}
    {%
      \item
      \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
        \small
        \textbf{#1}%
        \ifblank{#2}{}{\CVEntrySeparator #2}%
        & \small\texttt{#3}\\
      \end{tabular*}%
    }%
}
